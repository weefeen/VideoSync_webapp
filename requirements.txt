# VideoSync_webapp — video creation from music_line_extractor packages.
# No dependency on music_line_extractor or VideoScoreSync; the interface
# between them is a folder layout, not an import.

Flask>=3.0
python-dotenv>=1.0
Pillow>=10.0            # band raster handling, recolouring, tiling

# Vector bands. Digital packages ship .svg, which we rasterise at exactly
# the output resolution instead of resampling a fixed-size PNG. Optional:
# without it, .png/.jpg bands still render.
cairosvg>=2.7

# The server. Flask's own is single-threaded, reloads on a whim and says on
# every startup that it is not for production. gunicorn is what the systemd
# unit runs; `python run.py` still uses Flask's for development.
gunicorn>=21.2

# The queue, when RABBITMQ_URL is set. Without it the app runs its queue in
# process and never imports this — so a developer machine needs no broker.
pika>=1.3

# Reads the geolocation database in deploy/fetch-geoip.sh, so a visitor's
# address becomes a country and a city without leaving this machine. Optional
# at runtime: without it the visitor list still renders, unresolved.
maxminddb>=2.6

# ffmpeg and ffprobe are external binaries, not Python packages.
# Windows: https://www.gyan.dev/ffmpeg/builds/
# Debian/Ubuntu: sudo apt install ffmpeg
